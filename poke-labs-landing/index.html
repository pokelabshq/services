Quick Start

Clone, run, done

git clone https://github.com/pokelabshq/services.git
cd services

# Start all services at once
for d in */server.py; do nohup python3 "$d" > "/tmp/$(dirname $d).log" 2>&1 & done

# Or start individual ones
nohup python3 color-api/server.py &    # Port 8767
nohup python3 link-preview/server.py &  # Port 8765
nohup python3 dashboard/server.py &     # Port 8780

# Test
curl http://localhost:8767/api/random
